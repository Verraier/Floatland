cmake_minimum_required(VERSION 3.22)
project(floatland_gtk C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This opt-in GTK4 integration requires Linux")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk4 gtk4-wayland libadwaita-1)
add_library(floatland-gtk SHARED controls.c drawing.cpp)
target_link_libraries(floatland-gtk PRIVATE PkgConfig::GTK m)
target_compile_features(floatland-gtk PRIVATE cxx_std_17)
target_compile_options(floatland-gtk PRIVATE -Wall -Wextra $<$<COMPILE_LANGUAGE:C>:-Werror=implicit-function-declaration>)
