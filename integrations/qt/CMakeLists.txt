cmake_minimum_required(VERSION 3.22)
project(FloatlandQtDecoration LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This optional integration requires Linux and Qt Wayland development headers.")
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Network WaylandClient)
# Newer Qt releases expose private targets as separately loaded packages.
# A missing imported target alone does not mean the headers are absent.
if(NOT TARGET Qt6::GuiPrivate)
    find_package(Qt6GuiPrivate ${Qt6Gui_VERSION} EXACT CONFIG REQUIRED)
endif()
if(NOT TARGET Qt6::WaylandClientPrivate)
    find_package(Qt6WaylandClientPrivate ${Qt6WaylandClient_VERSION} EXACT CONFIG REQUIRED)
endif()
message(STATUS "Floatland Qt integration: Qt ${Qt6_VERSION}; private targets loaded")
set(CMAKE_AUTOMOC ON)
add_library(floatland-qt MODULE decoration.cpp)
target_compile_features(floatland-qt PRIVATE cxx_std_17)
target_link_libraries(floatland-qt PRIVATE Qt6::Core Qt6::Network Qt6::GuiPrivate Qt6::WaylandClientPrivate)
set_target_properties(floatland-qt PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/wayland-decoration-client")
