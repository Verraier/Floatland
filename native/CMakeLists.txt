cmake_minimum_required(VERSION 3.22)
project(floatland_bars VERSION 0.6.10 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Build native titlebars on the Linux host running Hyprland 0.56.2")
endif()
find_package(PkgConfig REQUIRED)
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_sdk.py"
    RESULT_VARIABLE sdk_result OUTPUT_VARIABLE sdk_output ERROR_VARIABLE sdk_error)
if(NOT sdk_result EQUAL 0)
    message(FATAL_ERROR "${sdk_error}")
endif()
message(STATUS "${sdk_output}")
pkg_check_modules(Native REQUIRED IMPORTED_TARGET hyprland=0.56.2 pixman-1 libdrm libinput libudev wayland-server xkbcommon pangocairo librsvg-2.0)
add_library(floatland-bars MODULE main.cpp titlebar.cpp)
set_target_properties(floatland-bars PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON PREFIX "")
target_link_libraries(floatland-bars PRIVATE PkgConfig::Native)
target_compile_options(floatland-bars PRIVATE -Wall -Wextra -Wno-unused-parameter)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(floatland-bars PRIVATE -fno-gnu-unique)
endif()
install(TARGETS floatland-bars LIBRARY DESTINATION lib/floatland)
enable_testing()
add_executable(titlebar-geometry-test geometry_test.cpp)
target_compile_features(titlebar-geometry-test PRIVATE cxx_std_17)
add_test(NAME titlebar-geometry COMMAND titlebar-geometry-test)

add_executable(overview-flow-test overview_flow_test.cpp)
target_compile_features(overview-flow-test PRIVATE cxx_std_17)
add_test(NAME overview-flow COMMAND overview-flow-test)

add_executable(switcher-layout-test switcher_layout_test.cpp)
target_compile_features(switcher-layout-test PRIVATE cxx_std_17)
add_test(NAME switcher-layout COMMAND switcher-layout-test)

add_executable(mini-state-test mini-state-test.cpp)
target_compile_features(mini-state-test PRIVATE cxx_std_17)
add_test(NAME mini-state COMMAND mini-state-test)
